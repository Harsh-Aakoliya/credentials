/*

 * CACHE MANAGER
 * ---------------------------------------------------
 */
class URLCache {
    private cache: Map<string, string> = new Map();

    get(instagramUrl: string): string | null {
        return this.cache.get(instagramUrl) || null;
    }

    set(instagramUrl: string, videoUrl: string): void {
        this.cache.set(instagramUrl, videoUrl);
    }

    has(instagramUrl: string): boolean {
        return this.cache.has(instagramUrl);
    }

    clear(): void {
        this.cache.clear();
    }

    size(): number {
        return this.cache.size;
    }
}

/*
 * ---------------------------------------------------
 * VIDEO COMPONENT WITH TAP TO PLAY/PAUSE
 * ---------------------------------------------------
 */
function ReelVideo({
    videoUrl,
    active,
    onProgress,
}: {
    videoUrl: string;
    active: boolean;
    onProgress?: (progress: number, duration: number) => void;
}) {
    const [isPlaying, setIsPlaying] = useState(true);
    const player = useVideoPlayer(videoUrl, (player) => {
        player.loop = true;
    });

    // Handle play/pause based on active state
    useEffect(() => {
        if (active && isPlaying) {
            player.play();
        } else {
            player.pause();
        }
    }, [active, isPlaying, player]);

    // Listen to player status for progress
    useEffect(() => {
        const interval = setInterval(() => {
            if (player && onProgress) {
                const currentTime = player.currentTime || 0;
                const duration = player.duration || 0;
                onProgress(currentTime, duration);
            }
        }, 100);

        return () => clearInterval(interval);
    }, [player, onProgress]);

    const togglePlayPause = () => {
        setIsPlaying((prev) => !prev);
    };

    return (
        <Pressable onPress={togglePlayPause} style={styles.videoContainer}>
            <VideoView
                player={player}
                style={styles.video}
                contentFit="cover"
                nativeControls={false}
            />
            
            {/* Play/Pause Indicator */}
            {!isPlaying && (
                <View style={styles.pauseOverlay}>
                    <View style={styles.pauseIcon}>
                        <Text style={styles.pauseIconText}>▶</Text>
                    </View>
                </View>
            )}
        </Pressable>
    );
}

/*
 * ---------------------------------------------------
 * TIMELINE COMPONENT
 * ---------------------------------------------------
 */
function Timeline({
    progress,
    duration,
    onSeek,
}: {
    progress: number;
    duration: number;
    onSeek?: (time: number) => void;
}) {
    const progressPercent = duration > 0 ? (progress / duration) * 100 : 0;

    return (
        <View style={styles.timelineContainer}>
            <View style={styles.timelineBackground}>
                <View
                    style={[
                        styles.timelineProgress,
                        { width: `${progressPercent}%` },
                    ]}
                />
            </View>
            <View style={styles.timeTextContainer}>
                <Text style={styles.timeText}>
                    {formatTime(progress)} / {formatTime(duration)}
                </Text>
            </View>
        </View>
    );
}

function formatTime(seconds: number): string {
    const mins = Math.floor(seconds / 60);
    const secs = Math.floor(seconds % 60);
    return `${mins}:${secs.toString().padStart(2, "0")}`;
}

/*
 * ---------------------------------------------------
 * MAIN APP
 * ---------------------------------------------------
 */
export default function Index() {
    const [category, setCategory] = useState<Category | null>(null);
    const [reels, setReels] = useState<Reel[]>([]);
    const [loading, setLoading] = useState(false);
    const [currentIndex, setCurrentIndex] = useState(0);
    const [progress, setProgress] = useState(0);
    const [duration, setDuration] = useState(0);

    /*
     * ---------------------------------------------------
     * RESOLVE INSTAGRAM URLS WITH CACHING
     * ---------------------------------------------------
     */
    const loadReels = async (selected: Category) => {
        setLoading(true);
        setReels([]);
        setCurrentIndex(0);

        const urls = data[selected];
        const resolved: Reel[] = [];

        for (const instagramUrl of urls) {
            try {
                // Check cache first
                if (urlCache.has(instagramUrl)) {
                    const cachedVideoUrl = urlCache.get(instagramUrl);
                    console.log("✓ Cache hit:", instagramUrl);
                    
                    if (cachedVideoUrl) {
                        resolved.push({
                            instagramUrl,
                            videoUrl: cachedVideoUrl,
                        });
                    }
                    continue;
                }

                // Not in cache, fetch from server
                console.log("Resolving:", instagramUrl);

                const response = await fetch(
                    `${SERVER}/resolve?url=${encodeURIComponent(instagramUrl)}`
                );

                if (!response.ok) {
                    console.log("Server returned:", response.status);
                    continue;
                }

                const result = await response.json();
                console.log("Resolve response:", result);

                if (result.videoUrl) {
                    // Store in cache
                    urlCache.set(instagramUrl, result.videoUrl);
                    console.log("✓ Cached:", instagramUrl);

                    resolved.push({
                        instagramUrl,
                        videoUrl: result.videoUrl,
                    });
                }
            } catch (error) {
                console.log("Failed to resolve:", instagramUrl, error);
            }
        }

        console.log(`Cache size: ${urlCache.size()} entries`);
        setReels(resolved);
        setLoading(false);
    };

    /*
     * ---------------------------------------------------
     * OPEN CATEGORY
     * ---------------------------------------------------
     */
    const openCategory = (selected: Category) => {
        setCategory(selected);
        loadReels(selected);
    };

    /*
     * ---------------------------------------------------
     * HANDLE VIDEO PROGRESS
     * ---------------------------------------------------
     */
    const handleProgress = (currentTime: number, totalDuration: number) => {
        setProgress(currentTime);
        setDuration(totalDuration);
    };

    /*
     * ---------------------------------------------------
     * CATEGORY SCREEN
     * ---------------------------------------------------
     */
    if (!category) {
        return (
            <View style={styles.home}>
                <Text style={styles.title}>My Reels</Text>
                
                <View style={styles.cacheInfo}>
                    <Text style={styles.cacheText}>
                        📦 Cached videos: {urlCache.size()}
                    </Text>
                    {urlCache.size() > 0 && (
                        <TouchableOpacity
                            onPress={() => {
                                urlCache.clear();
                                console.log("Cache cleared");
                            }}
                        >
                            <Text style={styles.clearCache}>Clear Cache</Text>
                        </TouchableOpacity>
                    )}
                </View>

                {(Object.keys(data) as Category[]).map((item) => (
                    <TouchableOpacity
                        key={item}
                        style={styles.category}
                        onPress={() => openCategory(item)}
                    >
                        <Text style={styles.categoryTitle}>
                            {item.toUpperCase()}
                        </Text>
                        <Text style={styles.categoryCount}>
                            {data[item].length} reels
                        </Text>
                    </TouchableOpacity>
                ))}
            </View>
        );
    }

    /*
     * ---------------------------------------------------
     * LOADING SCREEN
     * ---------------------------------------------------
     */
    if (loading) {
        return (
            <View style={styles.loading}>
                <ActivityIndicator size="large" color="#fff" />
                <Text style={styles.loadingText}>Loading reels...</Text>
                <TouchableOpacity
                    style={styles.loadingBack}
                    onPress={() => setCategory(null)}
                >
                    <Text style={styles.backText}>← Back</Text>
                </TouchableOpacity>
            </View>
        );
    }

    /*
     * ---------------------------------------------------
     * FEED
     * ---------------------------------------------------
     */
    return (
        <View style={styles.feed}>
            <FlatList
                data={reels}
                keyExtractor={(item) => item.instagramUrl}
                pagingEnabled
                showsVerticalScrollIndicator={false}
                decelerationRate="fast"
                snapToInterval={height}
                snapToAlignment="start"
                onMomentumScrollEnd={(event) => {
                    const index = Math.round(
                        event.nativeEvent.contentOffset.y / height
                    );
                    setCurrentIndex(index);
                    setProgress(0);
                    setDuration(0);
                }}
                renderItem={({ item, index }) => (
                    <View style={styles.reelContainer}>
                        <ReelVideo
                            videoUrl={item.videoUrl}
                            active={index === currentIndex}
                            onProgress={
                                index === currentIndex
                                    ? handleProgress
                                    : undefined
                            }
                        />

                        {/* Back button */}
                        <Pressable
                            style={styles.backButton}
                            onPress={() => setCategory(null)}
                        >
                            <Text style={styles.backArrow}>←</Text>
                        </Pressable>

                        {/* Category label */}
                        <View style={styles.categoryLabel}>
                            <Text style={styles.categoryLabelText}>
                                {category}
                            </Text>
                        </View>

                        {/* Timeline */}
                        {index === currentIndex && (
                            <Timeline progress={progress} duration={duration} />
                        )}

                        {/* Reel counter */}
                        <View style={styles.reelCounter}>
                            <Text style={styles.reelCounterText}>
                                {index + 1} / {reels.length}
                            </Text>
                        </View>
                    </View>
                )}
            />

            {reels.length === 0 && (
                <View style={styles.empty}>
                    <Text style={styles.emptyText}>
                        No playable reels found
                    </Text>
                    <TouchableOpacity onPress={() => setCategory(null)}>
                        <Text style={styles.goBack}>← Go back</Text>
                    </TouchableOpacity>
                </View>
            )}
        </View>
    );
}
